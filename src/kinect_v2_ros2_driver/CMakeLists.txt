cmake_minimum_required(VERSION 3.16)
project(kinect_v2_ros2_driver)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(PkgConfig REQUIRED)
pkg_check_modules(FREENECT2 REQUIRED IMPORTED_TARGET freenect2)
add_executable(kinect_publisher src/kinect_publisher.cpp)
target_compile_features(kinect_publisher PUBLIC cxx_std_17)
ament_target_dependencies(kinect_publisher rclcpp sensor_msgs)
target_link_libraries(kinect_publisher PkgConfig::FREENECT2)
install(TARGETS kinect_publisher DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY launch config DESTINATION share/${PROJECT_NAME})
ament_package()
